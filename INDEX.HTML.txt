    let currentTrack = 'c';
    let currentLessonIdx = 0;

    function selectTrack(track) {
      currentTrack = track;
      currentLessonIdx = 0;

      // Update button visual states
      const cBtn = document.getElementById('track-btn-c');
      const javaBtn = document.getElementById('track-btn-java');
      const viewer = document.getElementById('lessonViewer');

      if (track === 'c') {
        cBtn.classList.add('active');
        javaBtn.classList.remove('active');
        viewer.classList.remove('java-active');
      } else {
        javaBtn.classList.add('active');
        cBtn.classList.remove('active');
        viewer.classList.add('java-active');
      }

      // Update Compiler Frame
      document.getElementById('ideFrame').src = courseData[track].compilerUrl;

      // Render module list
      renderSidebar();
      renderCurrentLesson();
    }

    function renderSidebar() {
      const trackData = courseData[currentTrack];
      document.getElementById('sidebarTitle').textContent = trackData.title;
      const listContainer = document.getElementById('moduleList');
      listContainer.innerHTML = '';

      trackData.lessons.forEach((lesson, index) => {
        const btn = document.createElement('button');
        btn.className = `lesson-nav-btn ${index === currentLessonIdx ? 'active' : ''}`;
        btn.innerHTML = `<span>${lesson.title}</span>`;
        btn.onclick = () => {
          currentLessonIdx = index;
          renderSidebar();
          renderCurrentLesson();
        };
        listContainer.appendChild(btn);
      });
    }

    function renderCurrentLesson() {
      const lesson = courseData[currentTrack].lessons[currentLessonIdx];
      document.getElementById('lessonTitle').textContent = lesson.title;
      document.getElementById('lessonDesc').textContent = lesson.description;
      document.getElementById('codeBlock').textContent = lesson.code;
      document.getElementById('codeLangTag').textContent = currentTrack === 'c' ? 'C Code' : 'Java Code';

      // Load quiz
      const quiz = lesson.quiz;
      document.getElementById('quizQuestion').textContent = quiz.question;
      const optsContainer = document.getElementById('quizOptions');
      optsContainer.innerHTML = '';
      const feedback = document.getElementById('quizFeedback');
      feedback.textContent = '';

      quiz.options.forEach((opt, idx) => {
        const optBtn = document.createElement('button');
        optBtn.className = 'quiz-option';
        optBtn.textContent = opt;
        optBtn.onclick = () => handleQuizAnswer(idx, quiz.correct, quiz.explanation, optBtn);
        optsContainer.appendChild(optBtn);
      });
    }

    function handleQuizAnswer(selectedIdx, correctIdx, explanation, clickedElement) {
      const allOptions = document.querySelectorAll('.quiz-option');
      allOptions.forEach(btn => {
        btn.disabled = true;
        btn.classList.remove('correct', 'wrong');
      });

      const feedback = document.getElementById('quizFeedback');
      if (selectedIdx === correctIdx) {
        clickedElement.classList.add('correct');
        feedback.style.color = 'var(--success)';
        feedback.textContent = '✓ Correct! ' + explanation;
      } else {
        clickedElement.classList.add('wrong');
        allOptions[correctIdx].classList.add('correct');
        feedback.style.color = 'var(--danger)';
        feedback.textContent = '✕ Incorrect. ' + explanation;
      }
    }

    function copyCode() {
      const code = document.getElementById('codeBlock').innerText;
      navigator.clipboard.writeText(code).then(() => {
        const copyBtn = document.querySelector('.copy-btn');
        copyBtn.textContent = 'Copied!';
        setTimeout(() => copyBtn.textContent = 'Copy Code', 2000);
      });
    }

    // Initialize default view
    selectTrack('c');
